cmake_minimum_required(VERSION 3.15)
project(TacCommsFilterChecks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(RENAMENOISE_DEMO_EXECUTABLE OFF)
set(RENAMENOISE_BENCHMARK_EXECUTABLE OFF)
add_subdirectory(inspect/opus-1.5.2 opus)
add_subdirectory(inspect/mumble-1.5.735/3rdparty/renamenoise renamenoise)
add_executable(taccomms-filter-test test-filter.cpp TacCommsDenoiser.cpp)
target_link_libraries(taccomms-filter-test PRIVATE opus renamenoise)
include(modern-rnnoise.cmake)
add_executable(taccomms-modern-test test-filter.cpp TacCommsDenoiser.cpp)
target_compile_definitions(taccomms-modern-test PRIVATE TACCOMMS_MODERN_RNNOISE)
target_link_libraries(taccomms-modern-test PRIVATE opus taccomms-rnnoise)
if(MSVC)
    target_compile_options(taccomms-filter-test PRIVATE /O2)
    target_compile_options(renamenoise PRIVATE /O2)
else()
    target_link_libraries(taccomms-filter-test PRIVATE m)
endif()
